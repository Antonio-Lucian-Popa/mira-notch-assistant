cmake_minimum_required(VERSION 3.22)
project(appmodules)
include(${REACT_ANDROID_DIR}/cmake-utils/ReactNative-application.cmake)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Portable Android CPU backend" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory(${MIRA_WHISPER_SOURCE} ${CMAKE_CURRENT_BINARY_DIR}/whisper-build EXCLUDE_FROM_ALL)
add_library(mira_whisper SHARED voice/whisper_jni.cpp)
target_compile_features(mira_whisper PRIVATE cxx_std_17)
target_link_libraries(mira_whisper PRIVATE whisper android log)
target_link_options(mira_whisper PRIVATE "-Wl,-z,max-page-size=16384")
